# UI 扩展

HotCRM UI 元数据位于 `src/views/`、`src/pages/`、`src/dashboards/`、`src/reports/` 和 `src/actions/`。

## 新增动作

```ts
// src/actions/opportunity.actions.ts
import type { Action } from '@objectstack/spec/ui';

export const MarkOpportunityReviewedAction: Action = {
  name: 'mark_opportunity_reviewed',
  label: 'Mark Reviewed',
  objectName: 'crm_opportunity',
  icon: 'check-circle',
  type: 'modal',
  target: 'mark_opportunity_reviewed',
  locations: ['record_header'],
  body: {
    language: 'js',
    capabilities: ['api.write'],
    source: `
      await ctx.api.object('crm_opportunity').update(input.id, {
        next_step: 'Reviewed by manager'
      });
      return { id: input.id };
    `,
  },
  refreshAfter: true,
};
```

## 新增视图和页面

- 列表和看板定义放在 `src/views/`。
- 记录页面和工具页面放在 `src/pages/`。
- 从对应的 `index.ts` 导出新元数据。
- 将用户可见标签加入 `src/translations/{locale}.ts`。
